   ```typescript
     buildExplainPrompt,
     EXPLAIN_SYSTEM_PROMPT,

   ```

2. Create explainCommand:
   ```typescript
   export const explainCommand = new Command('explain')
     .description('Explain existing Ansible code in plain English')
     .argument('<path>', 'Path to Ansible file or role directory')
     .option('--playbook <path>', 'Provide context from a playbook or role')
     .option('--complex', 'Use Claude Opus for deeper analysis (higher cost)')
     .option('-q, --quiet', 'Suppress progress output')
     .action(async (ansiblePath: string, options: ExplainOptions) => {
       // Implementation below
     });
   ```

3. Implement action handler:
   a. Load config and validate API key (same pattern as new.ts)
   b. Call selectModel(options.complex) - if declined, return early
   c. Check if path exists with stat(), error if not found
   d. Create phase tracker: createPhaseTracker(options.quiet ?? false)

   e. Read the ansible path:
      - tracker.start('Reading Ansible code...')
      - const files = await readAnsiblePath(ansiblePath)
      - Determine fileType based on whether it's a directory ('role') or single file
      - tracker.succeed('Code loaded')

   f. Extract context if --playbook provided:
      - if (options.playbook) {
          tracker.start('Extracting context...')
          context = await extractContext(options.playbook)
          tracker.succeed('Context extracted')
        }

   g. Build content string:
      - If single file: content = files[0].content
      - If role directory: concatenate files with headers:
        ```
        === ${file.type}/${filename} ===
        ${file.content}
        ```

   h. Build prompt and stream response:
      - const prompt = buildExplainPrompt(content, fileType, context)
      - const client = createClient({ apiKey: config.api.key })
      - console.log(chalk.cyan('\n=== Explanation ===\n'))
      - Use streamMessage() with model from selectModel result
      - Capture full response for confidence check

   i. After streaming completes:
      - If not using --complex, call suggestComplexIfNeeded(fullResponse, options.complex)

   j. Error handling:
      - Wrap in try/catch
      - Transform API errors with transformApiError and display with displayApiError
      - Handle file not found errors gracefully

4. Define ExplainOptions interface:
   ```typescript
   interface ExplainOptions {
     playbook?: string;
     complex?: boolean;
     quiet?: boolean;
   }
   ```

Follow existing patterns from new.ts for error handling and progress tracking.
