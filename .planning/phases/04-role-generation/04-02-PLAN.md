```typescript
/** Galaxy-standard role directories */
export const ROLE_DIRECTORIES = [
  'tasks',
  'handlers',
  'defaults',
  'vars',
  'templates',
  'files',
  'meta',
  'molecule/default',
] as const;

/** Required files that must exist in a valid role */
export const REQUIRED_FILES = [
  'tasks/main.yml',
  'defaults/main.yml',
  'meta/main.yml',
  'README.md',
] as const;

export interface RoleStructureOptions {
  roleName: string;
  outputDir: string;
  dryRun?: boolean;
}

/**
 * Create the role directory structure.
 * Creates all directories with .gitkeep files for empty ones.
 */
export async function createRoleStructure(options: RoleStructureOptions): Promise<string[]>
```
- Create role root directory: `{outputDir}/{roleName}`
- Create all ROLE_DIRECTORIES
- Add `.gitkeep` to empty directories (files/, templates/ when no content)
- Return list of created directories
- Respect dryRun option (don't actually create, just return what would be created)

**parser.ts** - Parse generated output with file markers:

```typescript
export interface GeneratedFile {
  path: string;      // e.g., "tasks/main.yml"
  content: string;   // YAML or text content
}

/**
 * Parse generated content using file markers.
 *
 * Format:
 * === PATH: tasks/main.yml ===
 * ---
 * [content]
 * === END ===
 *
 * @param output - Raw generated output from Claude
 * @returns Array of parsed files
 */
export function parseGeneratedFiles(output: string): GeneratedFile[]
```
- Use regex: `/=== PATH: (.+?) ===([\s\S]*?)=== END ===/g`
- Trim whitespace from path and content
- Handle missing files gracefully (return empty array, log warning)
- Skip malformed blocks (log warning but don't fail)

**validator.ts** - YAML validation:

```typescript
import { parse } from 'yaml';

export interface ValidationResult {
  valid: boolean;
  error?: string;
  path?: string;
}

/**
 * Validate a single YAML string.
 * @param content - YAML content to validate
 * @param path - File path for error messages
 */
export function validateYaml(content: string, path: string): ValidationResult

/**
 * Validate all generated files.
 * Only validates .yml/.yaml files.
 * @param files - Generated files to validate
 * @returns Array of validation results (only failures)
 */
export function validateAllFiles(files: GeneratedFile[]): ValidationResult[]

/**
 * Check for FQCN usage in tasks.
 * Warns if short module names detected (apt instead of ansible.builtin.apt).
 * @param content - YAML content
 * @returns Array of warnings (non-blocking)
 */
export function checkFqcn(content: string): string[]

/**
 * Check for basic idempotency patterns.
 * Warns if state: parameter missing from stateful modules.
 * @param content - YAML content
 * @returns Array of warnings (non-blocking)
 */
export function checkIdempotency(content: string): string[]
```

**FQCN check** - Look for common short names without `ansible.builtin.` or `community.` prefix:
- `apt:`, `yum:`, `dnf:`, `package:` without prefix
- `file:`, `copy:`, `template:` without prefix
- `service:`, `systemd:` without prefix
- Use regex to avoid false positives inside strings

**Idempotency check** - Look for modules that should have state:
- `ansible.builtin.apt`, `ansible.builtin.dnf`, `ansible.builtin.package` -> state: present/absent
- `ansible.builtin.service`, `ansible.builtin.systemd_service` -> state: started/stopped
- `ansible.builtin.file` -> state: file/directory/absent/link
- Warn if module used but state not found in same task block
