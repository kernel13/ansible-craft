```typescript
import Anthropic from '@anthropic-ai/sdk';
import ora from 'ora';
import { DEFAULT_MODEL } from '../ai/client.js';
import { withRetry } from '../ai/retry.js';
import { transformApiError, displayApiError } from '../ai/errors.js';
import { ANSIBLE_EXPERT_SYSTEM_PROMPT, buildPlanPrompt } from './prompts/index.js';
import { PLAN_PREVIEW_SCHEMA, type PlanPreview } from './schemas/plan-preview.js';

export interface GenerateOptions {
  /** Skip interactive clarifying questions */
  noInteractive?: boolean;
  /** Custom output directory */
  outputDir?: string;
  /** Override inferred role name */
  roleName?: string;
  /** Preview without writing files */
  dryRun?: boolean;
  /** Overwrite existing directory */
  force?: boolean;
  /** Suppress output */
  quiet?: boolean;
  /** No retries on failure */
  noRetry?: boolean;
}

/**
 * Generate a role plan preview using structured outputs.
 *
 * Uses Anthropic's structured-outputs-2025-11-13 beta to guarantee
 * the response matches our PlanPreview schema.
 *
 * @param client - Configured Anthropic client
 * @param description - Natural language role description
 * @param clarifications - Optional answers to clarifying questions
 * @param options - Generation options
 * @returns Validated plan preview
 */
export async function generateRolePlan(
  client: Anthropic,
  description: string,
  clarifications?: Record<string, string>,
  options: Pick<GenerateOptions, 'quiet' | 'noRetry'> = {},
): Promise<PlanPreview>
```

**Implementation details:**

1. **Show spinner** (unless quiet): "Generating role plan..."

2. **Build the prompt** using `buildPlanPrompt(description, clarifications)`

3. **Call Anthropic beta API** with structured outputs:
```typescript
const response = await withRetry(async () => {
  return client.beta.messages.create({
    model: DEFAULT_MODEL,
    max_tokens: 4096,
    betas: ['structured-outputs-2025-11-13'],
    system: ANSIBLE_EXPERT_SYSTEM_PROMPT,
    messages: [{ role: 'user', content: prompt }],
    response_format: {
      type: 'json_schema',
      json_schema: {
        name: 'plan_preview',
        schema: PLAN_PREVIEW_SCHEMA,
        strict: true,
      },
    },
  });
}, { noRetry: options.noRetry, quiet: options.quiet });
```

4. **Parse the JSON response**:
```typescript
const textBlock = response.content.find(b => b.type === 'text');
if (!textBlock || textBlock.type !== 'text') {
  throw new Error('No text content in response');
}
const plan = JSON.parse(textBlock.text) as PlanPreview;
```

5. **Stop spinner and return plan**

6. **Handle errors** - wrap in try/catch, use `transformApiError` and `displayApiError` for user-friendly messages

Note: The structured outputs beta guarantees the JSON matches our schema, so we don't need additional validation.

```typescript
 * @param client - Configured Anthropic client
 * @param plan - Approved plan preview

 * @param options - Generation options
 * @returns Parsed generated files
 */
export async function generateRoleCode(

  plan: PlanPreview,
  description: string,
  options: Pick<GenerateOptions, 'quiet' | 'noRetry'> = {},
): Promise<GeneratedFile[]>
```

**Implementation details:**

1. **Build generation prompt** using `buildGeneratePrompt(plan, description)`

2. **Use streamMessage** for visual streaming:
```typescript
const message = await streamMessage(client, {
  userMessage: prompt,
  systemPrompt: ANSIBLE_EXPERT_SYSTEM_PROMPT,
  maxTokens: 8192,  // Roles can be large
}, {
  quiet: options.quiet,
  noRetry: options.noRetry,
});
```

Note: `streamMessage` already handles:
- "Connecting to Claude..." spinner
- Spinner stop on first token
- Streaming tokens to stdout
- Retry logic with exponential backoff
- Error handling with newline cleanup

3. **Extract and parse output**:

4. **Validate file count** - warn if fewer files than expected based on plan

5. **Return parsed files** for validation and writing
