```typescript
/**
 * Integration tests for wizard CLI integration in new command.
 *
 * Tests wizard invocation conditions, skip conditions, and error handling.
 */

import { afterEach, beforeEach, describe, expect, mock, spyOn, test } from 'bun:test';

// Mock process.stdin.isTTY before importing modules
const originalIsTTY = process.stdin.isTTY;

// Mock the wizard modules
const mockRunRoleWizard = mock(() =>
  Promise.resolve({
    structure: ['tasks', 'handlers'],
    platforms: ['Ubuntu'],
    handlers: ['restart'],
    custom: {},
  }),
);

const mockRunPlaybookWizard = mock(() =>
  Promise.resolve({
    hosts: ['webservers'],
    become: true,
    includeHandlers: true,
    custom: {},
  }),
);

// Mock ExitPromptError
class MockExitPromptError extends Error {
  constructor() {
    super('User cancelled');
    this.name = 'ExitPromptError';
  }
}

mock.module('../../wizard/role-wizard.js', () => ({
  runRoleWizard: mockRunRoleWizard,
}));

mock.module('../../wizard/playbook-wizard.js', () => ({
  runPlaybookWizard: mockRunPlaybookWizard,
}));

mock.module('@inquirer/core', () => ({
  ExitPromptError: MockExitPromptError,
}));

// Mock inquirer prompts to prevent actual prompts during tests
mock.module('@inquirer/prompts', () => ({
  select: mock(() => Promise.resolve('accept')),
  input: mock(() => Promise.resolve('')),
  confirm: mock(() => Promise.resolve(true)),
}));

// Mock AI client and generation
mock.module('../../ai/client.js', () => ({
  createClient: mock(() => ({})),
  DEFAULT_MODEL: 'claude-sonnet-4-5-20250929',
}));

mock.module('../../config/index.js', () => ({
  loadConfig: mock(() =>
    Promise.resolve({
      api: { key: 'test-key' },
    }),
  ),
}));

mock.module('../../generation/index.js', () => ({
  generateRolePlan: mock(() =>
    Promise.resolve({
      role_name: 'test',
      description: 'Test role',
      tasks: [],
      variables: [],
      handlers: [],
      templates: [],
      platforms: [],
    }),
  ),
  generatePlaybookPlan: mock(() =>
    Promise.resolve({
      playbook_name: 'test',
      description: 'Test playbook',
      inventory_groups: ['all'],
      plays: [],
      group_vars: [],
    }),
  ),
  generateRoleCode: mock(() => Promise.resolve([])),
  generatePlaybookCode: mock(() => Promise.resolve([])),
  validateGeneratedFiles: mock(() => ({ valid: true, errors: [], warnings: [] })),
  isAnsibleLintAvailable: mock(() => Promise.resolve(false)),
  writeGeneratedRole: mock(() => Promise.resolve({ roleDir: '/tmp/test', filesWritten: [] })),
  writeGeneratedPlaybook: mock(() =>
    Promise.resolve({ playbookDir: '/tmp/test', filesWritten: [] }),
  ),
  inferRoleName: mock(() => 'test-role'),
  inferPlaybookName: mock(() => 'test-playbook'),
  sanitizeRoleName: mock((name: string) => name),
  displayRoleTree: mock(() => {}),
  displayPlaybookTree: mock(() => {}),
  displayValidationReport: mock(() => {}),
  formatInstallInstructions: mock(() => ''),
}));

// Import after mocks
import { newCommand } from './new.js';
import * as generationModule from '../../generation/index.js';

describe('wizard CLI integration', () => {
  let consoleLogSpy: ReturnType<typeof spyOn>;
  let consoleErrorSpy: ReturnType<typeof spyOn>;
  let processExitSpy: ReturnType<typeof spyOn>;

  beforeEach(() => {
    // Reset mocks
    mockRunRoleWizard.mockClear();
    mockRunPlaybookWizard.mockClear();

    // Suppress console output
    consoleLogSpy = spyOn(console, 'log').mockImplementation(() => {});
    consoleErrorSpy = spyOn(console, 'error').mockImplementation(() => {});
    processExitSpy = spyOn(process, 'exit').mockImplementation(() => {
      throw new Error('process.exit called');
    });

    // Default to TTY
    Object.defineProperty(process.stdin, 'isTTY', {
      value: true,
      writable: true,
      configurable: true,
    });
  });

  afterEach(() => {
    consoleLogSpy.mockRestore();
    consoleErrorSpy.mockRestore();
    processExitSpy.mockRestore();
    Object.defineProperty(process.stdin, 'isTTY', {
      value: originalIsTTY,
      writable: true,
      configurable: true,
    });
  });

  describe('role command wizard invocation', () => {
    test('calls wizard when TTY and no skip flags', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });

      await newCommand.parseAsync(['node', 'test', 'role', 'test role', '--dry-run', '--force']);

      expect(mockRunRoleWizard).toHaveBeenCalled();
    });

    test('skips wizard with --quick flag', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });

      await newCommand.parseAsync([
        'node',
        'test',
        'role',
        'test role',
        '--quick',
        '--dry-run',
        '--force',
      ]);

      expect(mockRunRoleWizard).not.toHaveBeenCalled();
    });

    test('skips wizard with --no-interactive flag', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });

      await newCommand.parseAsync([
        'node',
        'test',
        'role',
        'test role',
        '--no-interactive',
        '--dry-run',
        '--force',
      ]);

      expect(mockRunRoleWizard).not.toHaveBeenCalled();
    });

    test('skips wizard in non-TTY environment', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: false });

      await newCommand.parseAsync(['node', 'test', 'role', 'test role', '--dry-run', '--force']);

      expect(mockRunRoleWizard).not.toHaveBeenCalled();
    });

    test('skips wizard with --json flag', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });

      await newCommand.parseAsync([
        'node',
        'test',
        'role',
        'test role',
        '--json',
        '--dry-run',
        '--force',
      ]);

      expect(mockRunRoleWizard).not.toHaveBeenCalled();
    });

    test('passes wizard clarifications to generateRolePlan', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });
      const generateRolePlanSpy = spyOn(
        generationModule,
        'generateRolePlan',
      ).mockResolvedValueOnce({
        role_name: 'test',
        description: 'Test',
        tasks: [],
        variables: [],
        handlers: [],
        templates: [],
        platforms: [],
      });

      await newCommand.parseAsync(['node', 'test', 'role', 'test role', '--dry-run', '--force']);

      expect(generateRolePlanSpy).toHaveBeenCalled();
      const callArgs = generateRolePlanSpy.mock.calls[0];
      // Third argument should be clarifications
      expect(callArgs[2]).toBeDefined();
      expect(callArgs[2]).toHaveProperty('structure');
      expect(callArgs[2]).toHaveProperty('platforms');

      generateRolePlanSpy.mockRestore();
    });

    test('handles ExitPromptError gracefully', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });
      mockRunRoleWizard.mockRejectedValueOnce(new MockExitPromptError());

      await newCommand.parseAsync(['node', 'test', 'role', 'test role', '--dry-run', '--force']);

      expect(consoleLogSpy).toHaveBeenCalledWith(expect.stringContaining('cancelled'));
    });
  });

  describe('playbook command wizard invocation', () => {
    test('calls wizard when TTY and no skip flags', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });

      await newCommand.parseAsync([
        'node',
        'test',
        'playbook',
        'test playbook',
        '--dry-run',
        '--force',
      ]);

      expect(mockRunPlaybookWizard).toHaveBeenCalled();
    });

    test('skips wizard with --quick flag', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: true });

      await newCommand.parseAsync([
        'node',
        'test',
        'playbook',
        'test playbook',
        '--quick',
        '--dry-run',
        '--force',
      ]);

      expect(mockRunPlaybookWizard).not.toHaveBeenCalled();
    });

    test('skips wizard in non-TTY environment', async () => {
      Object.defineProperty(process.stdin, 'isTTY', { value: false });

      await newCommand.parseAsync([
        'node',
        'test',
        'playbook',
        'test playbook',
        '--dry-run',
        '--force',
      ]);

      expect(mockRunPlaybookWizard).not.toHaveBeenCalled();
    });
  });
});
```
